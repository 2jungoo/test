#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Node {
    char *line;
    struct Node *next;
} Node;

// 사전순 삽입 함수
void insert(Node **head, char *str) {
    Node *newNode = (Node *)malloc(sizeof(Node));
    newNode->line = strdup(str);  // malloc + strcpy 기능
    newNode->next = NULL;

    if (*head == NULL || strcmp(str, (*head)->line) < 0) {
        newNode->next = *head;
        *head = newNode;
        return;
    }

    Node *cur = *head;
    while (cur->next && strcmp(str, cur->next->line) > 0)
        cur = cur->next;

    newNode->next = cur->next;
    cur->next = newNode;
}

// 리스트 출력 함수
void printList(Node *head) {
    while (head) {
        printf("%s", head->line);
        head = head->next;
    }
}

// 메모리 해제 함수
void freeList(Node *head) {
    while (head) {
        Node *temp = head;
        head = head->next;
        free(temp->line);
        free(temp);
    }
}

int main() {
    Node *head = NULL;
    char buffer[256];

    printf("Enter lines (Ctrl+D to end):\n");
    while (fgets(buffer, sizeof(buffer), stdin)) {
        insert(&head, buffer);
    }

    printf("\n--- Sorted Lines ---\n");
    printList(head);
    freeList(head);
    return 0;
}
