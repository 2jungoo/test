#include <stdio.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>

int main(int argc, char *argv[]) {
    if (argc != 3) {
        fprintf(stderr, "Usage: %s <src> <dst>\n", argv[0]);
        return 1;
    }

    /* 원본 파일 열기 */
    int src = open(argv[1], O_RDONLY);
    if (src < 0) { perror("open src"); return 1; }

    /* 파일 상태 확인 */
    struct stat sb;
    if (fstat(src, &sb) == -1) { perror("stat"); return 1; }

    if (sb.st_size == 0) {
        fprintf(stderr, "Error: source file is 0 bytes\n");
        return 1;
    }

    /* 대상 파일 생성 */
    int dst = open(argv[2], O_RDWR | O_CREAT | O_TRUNC, 0644);
    if (dst < 0) { perror("open dst"); return 1; }

    /* 대상 파일 크기 설정 */
    if (ftruncate(dst, sb.st_size) == -1) {
        perror("ftruncate");
        return 1;
    }

    /* mmap */
    void *src_map = mmap(NULL, sb.st_size, PROT_READ, MAP_PRIVATE, src, 0);
    if (src_map == MAP_FAILED) { perror("mmap src"); return 1; }

    void *dst_map = mmap(NULL, sb.st_size, PROT_WRITE, MAP_SHARED, dst, 0);
    if (dst_map == MAP_FAILED) { perror("mmap dst"); return 1; }

    /* memcpy */
    memcpy(dst_map, src_map, sb.st_size);

    /* 정리 */
    munmap(src_map, sb.st_size);
    munmap(dst_map, sb.st_size);
    close(src);
    close(dst);

    printf("File copied successfully!\n");
    return 0;
}
